# Cloud Setup (Supabase)

## 环境变量 (Vite)
在 `.env.local` 写入：

```
VITE_SUPABASE_URL=你的项目URL
VITE_SUPABASE_ANON_KEY=你的Anon Key
```

## 数据库表 (SQL 摘要)
按顺序运行以下 SQL（Supabase SQL Editor）：

```sql
-- accounts
create table if not exists accounts (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null,
  name text not null,
  type text not null check (type in ('admin','parent','child')),
  created_at timestamptz default now()
);

-- profiles
create table if not exists profiles (
  account_id uuid primary key references accounts(id) on delete cascade,
  exp bigint default 0,
  level int default 1,
  streak int default 0,
  last_active_date date,
  total_questions int default 0,
  correct_questions int default 0,
  total_time_ms bigint default 0
);

-- daily_stats
create table if not exists daily_stats (
  account_id uuid references accounts(id) on delete cascade,
  date date not null,
  questions_answered int default 0,
  correct_answers int default 0,
  total_time_ms bigint default 0,
  exp_gained int default 0,
  primary key (account_id, date)
);

-- daily_tasks
create table if not exists daily_tasks (
  account_id uuid references accounts(id) on delete cascade,
  date date not null,
  task_key text not null,
  progress int default 0,
  completed boolean default false,
  primary key (account_id, date, task_key)
);

-- question_logs
create table if not exists question_logs (
  id bigint generated by default as identity primary key,
  account_id uuid references accounts(id) on delete cascade,
  created_at timestamptz default now(),
  display_text text,
  correct_answer int,
  user_answer int,
  is_correct boolean,
  time_taken numeric
);

-- wrong_questions
create table if not exists wrong_questions (
  id bigint generated by default as identity primary key,
  account_id uuid references accounts(id) on delete cascade,
  created_at timestamptz default now(),
  payload jsonb
);
```

## 行级安全 (RLS)
为上述表开启 RLS，并添加策略：

```sql
alter table accounts enable row level security;
alter table profiles enable row level security;
alter table daily_stats enable row level security;
alter table daily_tasks enable row level security;
alter table question_logs enable row level security;
alter table wrong_questions enable row level security;

create policy "owner_rw_accounts" on accounts
  using (auth.uid() = user_id) with check (auth.uid() = user_id);

create policy "owner_rw_profiles" on profiles
  using (exists(select 1 from accounts a where a.id = profiles.account_id and a.user_id = auth.uid()))
  with check (exists(select 1 from accounts a where a.id = profiles.account_id and a.user_id = auth.uid()));

create policy "owner_rw_daily_stats" on daily_stats
  using (exists(select 1 from accounts a where a.id = daily_stats.account_id and a.user_id = auth.uid()))
  with check (exists(select 1 from accounts a where a.id = daily_stats.account_id and a.user_id = auth.uid()));

create policy "owner_rw_daily_tasks" on daily_tasks
  using (exists(select 1 from accounts a where a.id = daily_tasks.account_id and a.user_id = auth.uid()))
  with check (exists(select 1 from accounts a where a.id = daily_tasks.account_id and a.user_id = auth.uid()));

create policy "owner_rw_question_logs" on question_logs
  using (exists(select 1 from accounts a where a.id = question_logs.account_id and a.user_id = auth.uid()))
  with check (exists(select 1 from accounts a where a.id = question_logs.account_id and a.user_id = auth.uid()));

create policy "owner_rw_wrong_questions" on wrong_questions
  using (exists(select 1 from accounts a where a.id = wrong_questions.account_id and a.user_id = auth.uid()))
  with check (exists(select 1 from accounts a where a.id = wrong_questions.account_id and a.user_id = auth.uid()));
```
